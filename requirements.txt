# Runtime dependencies (pinned). Audit with: pip-audit -r requirements.txt
#
# Requires Python >= 3.10. The floor is set by pdfminer.six: the release that
# fixes PYSEC-2026-1761 dropped 3.9, and pdfminer is the code that parses the
# untrusted contract PDF, so staying back was not an option. CI builds on 3.11.
#
# pdfplumber pins pdfminer.six exactly and constrains Pillow, so bumping it is
# how the parsing stack gets patched -- don't pin pdfminer/Pillow separately here
# or the two will fight.
pdfplumber==0.11.10

# Web UI / API (csms.webapp). Pure-Python, cross-platform.
Flask==3.1.3

# Playbook config (YAML). Optional at runtime — JSON playbooks need no extra dep.
PyYAML>=6,<7

# TLS trust for the Asana calls (csms.asana_client). The frozen app carries no
# CA bundle, and OpenSSL's default paths read neither the macOS Keychain nor the
# Windows certificate store -- so on a network with a filtering proxy the app
# fails to verify a root the user's browser accepts. truststore delegates to the
# OS store; certifi is the fallback when there is no OS store to delegate to.
truststore==0.10.4
certifi==2026.7.22
